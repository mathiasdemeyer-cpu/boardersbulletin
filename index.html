Automatische 20:00 WhatsApp-kanaal Bot

Met onderstaand Python-script en een gratis GitHub Actions schema ontvang je elke avond om 20:00 uur automatisch deze update in jullie WhatsApp-kanaal (via Whapi.Cloud) of rechtstreeks op je eigen WhatsApp (via CallMeBot).

1. update_ischgl_whatsapp.py 
import os
import requests
from datetime import datetime

# Coördinaten Silvretta Arena
LOCS = {
    "ischgl": {"lat": 47.0131, "lon": 10.2914, "elev": 1377},
    "samnaun": {"lat": 46.9442, "lon": 10.3614, "elev": 1846},
    "berg": {"lat": 46.9739, "lon": 10.3208, "elev": 2872},
}

def fetch_forecast():
    url = "https://api.open-meteo.com/v1/forecast"
    params = {
        "latitude": [LOCS["ischgl"]["lat"], LOCS["samnaun"]["lat"], LOCS["berg"]["lat"]],
        "longitude": [LOCS["ischgl"]["lon"], LOCS["samnaun"]["lon"], LOCS["berg"]["lon"]],
        "elevation": [1377, 1846, 2872],
        "current": ["temperature_2m", "snow_depth"],
        "hourly": ["temperature_2m", "snowfall", "freezing_level_height"],
        "models": "best_match,icon_seamless,ecmwf_ifs025",
        "forecast_days": 3,
        "timezone": "Europe/Vienna"
    }
    r = requests.get(url, params=params, timeout=20)
    r.raise_for_status()
    return r.json()

def build_message(data):
    ischgl, samnaun, berg = data[0], data[1], data[2]
    now_str = datetime.now().strftime("%d/%m/%Y")

    depth_berg = round((berg["current"].get("snow_depth") or 0) * 100)
    depth_sam = round((samnaun["current"].get("snow_depth") or 0) * 100)
    depth_isc = round((ischgl["current"].get("snow_depth") or 0) * 100)

    snow_24_berg = round(sum((berg["hourly"].get("snowfall_icon_seamless") or [0])[:24]), 1)
    snow_72_berg = round(sum((berg["hourly"].get("snowfall_icon_seamless") or [0])[:72]), 1)
    snow_72_ecmwf = round(sum((berg["hourly"].get("snowfall_ecmwf_ifs025") or [0])[:72]), 1)

    temps_berg = (berg["hourly"].get("temperature_2m_icon_seamless") or [0])[:24]
    temps_dal = (ischgl["hourly"].get("temperature_2m_icon_seamless") or [0])[:24]
    flh = (berg["hourly"].get("freezing_level_height_icon_seamless") or [1500])[:24]
    snowline = max(800, round(sum(flh) / len(flh) - 250))

    return (
        f"🏔️ *Silvretta Powder Radar — {now_str} (20:00)*\n"
        f"_Ischgl & Samnaun Update voor Mathias & Wouter_\n\n"
        f"❄️ *Huidig Sneeuwdek*\n"
        f"• Greitspitz (2.872m): *{depth_berg} cm* ({berg['current']['temperature_2m']}°C)\n"
        f"• Samnaun (1.846m): *{depth_sam} cm* ({samnaun['current']['temperature_2m']}°C)\n"
        f"• Ischgl (1.377m): *{depth_isc} cm* ({ischgl['current']['temperature_2m']}°C)\n\n"
        f"🌨️ *Verwachte Sneeuwval (Berg)*\n"
        f"• Komende 24u (ICON): *{snow_24_berg} cm*\n"
        f"• Komende 72u: *{snow_72_berg} cm* (ICON) / *{snow_72_ecmwf} cm* (ECMWF)\n\n"
        f"🌡️ *Temperatuur & Sneeuwgrens (24u)*\n"
        f"• Berg (2.872m): {min(temps_berg):.1f}°C tot {max(temps_berg):.1f}°C\n"
        f"• Vallei Ischgl (1.377m): {min(temps_dal):.1f}°C tot {max(temps_dal):.1f}°C\n"
        f"• Gem. sneeuwgrens: *~{snowline} m*"
    )

def send_whatsapp_channel(msg):
    # Optie A: WhatsApp Kanaal via Whapi.Cloud (NL-... @newsletter)
    whapi_token = os.getenv("WHAPI_TOKEN")
    channel_id = os.getenv("WHATSAPP_CHANNEL_ID")
    if whapi_token and channel_id:
        requests.post(
            "https://gate.whapi.cloud/messages/text",
            headers={"Authorization": f"Bearer {whapi_token}"},
            json={"to": channel_id, "body": msg},
            timeout=15
        )
        return

    # Optie B: Persoonlijk WhatsApp-bericht via CallMeBot
    phone = os.getenv("CALLMEBOT_PHONE")
    apikey = os.getenv("CALLMEBOT_APIKEY")
    if phone and apikey:
        requests.get(
            "https://api.callmebot.com/whatsapp.php",
            params={"phone": phone, "text": msg, "apikey": apikey},
            timeout=15
        )

if __name__ == "__main__":
    forecast = fetch_forecast()
    message = build_message(forecast)
    print(message)
    send_whatsapp_channel(message)
2. GitHub Actions (.github/workflows/snow-update.yml) 
name: Silvretta 20:00 WhatsApp Update
on:
  schedule:
    - cron: '0 19 * * *' # 19:00 UTC = 20:00 CET (Brussel/Ischgl)
  workflow_dispatch:

jobs:
  send-update:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: '3.11'
      - run: pip install requests
      - env:
          WHAPI_TOKEN: ${{ secrets.WHAPI_TOKEN }}
          WHATSAPP_CHANNEL_ID: ${{ secrets.WHATSAPP_CHANNEL_ID }}
          CALLMEBOT_PHONE: ${{ secrets.CALLMEBOT_PHONE }}
          CALLMEBOT_APIKEY: ${{ secrets.CALLMEBOT_APIKEY }}
        run: python update_ischgl_whatsapp.py
